-- 3. KATEGORI TABLOSU

(
    KategoriId INT PRIMARY KEY IDENTITY(1,1),
    KategoriAdi NVARCHAR(100) NOT NULL UNIQUE,
    Aciklama NVARCHAR(500) NULL,
    OlusturulmaTarihi DATETIME DEFAULT GETDATE(),
    GuncellemeTarihi DATETIME DEFAULT GETDATE()
);
PRINT 'Kategori tablosu olusturuldu.';
GO

-- 4. BIRIM TABLOSU
CREATE TABLE Birim
(
    BirimId INT PRIMARY KEY IDENTITY(1,1),
    BirimAdi NVARCHAR(50) NOT NULL UNIQUE,
    Kisaltma NVARCHAR(10) NOT NULL,
    OlusturulmaTarihi DATETIME DEFAULT GETDATE()
);
PRINT 'Birim tablosu olusturuldu.';
GO

-- 5. DEPO TABLOSU
CREATE TABLE Depo
(
    DepoId INT PRIMARY KEY IDENTITY(1,1),
    DepoAdi NVARCHAR(100) NOT NULL UNIQUE,
    Lokasyon NVARCHAR(200) NULL,
    Aciklama NVARCHAR(500) NULL,
    AktifMi BIT DEFAULT 1,
    OlusturulmaTarihi DATETIME DEFAULT GETDATE(),
    GuncellemeTarihi DATETIME DEFAULT GETDATE()
);
PRINT 'Depo tablosu olusturuldu.';
GO

-- 6. MALZEME KATALOGU VE URUN TABLOSU
CREATE TABLE MalzemeKatalog
(
    MalzemeId INT IDENTITY(1,1) NOT NULL CONSTRAINT PK_MalzemeKatalog PRIMARY KEY,
    MalzemeNo NVARCHAR(50) NOT NULL CONSTRAINT UQ_MalzemeKatalog_MalzemeNo UNIQUE,
    Barkod NVARCHAR(50) NOT NULL CONSTRAINT UQ_MalzemeKatalog_Barkod UNIQUE,
    Aciklama NVARCHAR(200) NOT NULL,
    BirimId INT NOT NULL,
    OlusturulmaTarihi DATETIME NOT NULL CONSTRAINT DF_MalzemeKatalog_OlusturulmaTarihi DEFAULT GETDATE(),
    CONSTRAINT FK_MalzemeKatalog_Birim FOREIGN KEY (BirimId) REFERENCES Birim(BirimId),
    CONSTRAINT CK_MalzemeKatalog_MalzemeNo CHECK (LEN(LTRIM(RTRIM(MalzemeNo))) > 0),
    CONSTRAINT CK_MalzemeKatalog_Aciklama CHECK (LEN(LTRIM(RTRIM(Aciklama))) > 0)
);
CREATE UNIQUE INDEX UX_MalzemeKatalog_Bag ON MalzemeKatalog(MalzemeId, Barkod, BirimId);
GO

CREATE TABLE Urun
(
    UrunId INT PRIMARY KEY IDENTITY(1,1),
    MalzemeId INT NOT NULL,
    Barkod NVARCHAR(50) NOT NULL UNIQUE,
    UrunAdi NVARCHAR(200) COLLATE Turkish_CI_AS NOT NULL,
    KategoriId INT NOT NULL,
    BirimId INT NOT NULL,
    Fiyat DECIMAL(18,2) NOT NULL DEFAULT 0,
    MinimumStok INT DEFAULT 10,
    MaksimumStok INT DEFAULT 1000,
    AktifMi BIT DEFAULT 1,
    OlusturulmaTarihi DATETIME DEFAULT GETDATE(),
    GuncellemeTarihi DATETIME DEFAULT GETDATE(),
    CONSTRAINT FK_Urun_Kategori FOREIGN KEY (KategoriId) REFERENCES Kategori(KategoriId),
    CONSTRAINT FK_Urun_Birim FOREIGN KEY (BirimId) REFERENCES Birim(BirimId),
    CONSTRAINT FK_Urun_MalzemeKatalog FOREIGN KEY (MalzemeId) REFERENCES MalzemeKatalog(MalzemeId),
    CONSTRAINT FK_Urun_MalzemeBag FOREIGN KEY (MalzemeId, Barkod, BirimId) REFERENCES MalzemeKatalog(MalzemeId, Barkod, BirimId),
    CONSTRAINT UQ_Urun_MalzemeId UNIQUE (MalzemeId),
    CONSTRAINT CK_Urun_Fiyat CHECK (Fiyat >= 0),
    CONSTRAINT CK_Urun_UrunAdi_Bosluk CHECK (DATALENGTH(UrunAdi) = DATALENGTH(LTRIM(RTRIM(UrunAdi))))
);
PRINT 'Urun tablosu olusturuldu.';
GO

-- Türkçe büyük/küçük harf eşitliğinde aynı ürün adı tekrar kaydedilemez.
CREATE UNIQUE INDEX UX_Urun_UrunAdi ON Urun(UrunAdi);
GO

-- 7. STOK TABLOSU
CREATE TABLE Stok
(
    StokId INT PRIMARY KEY IDENTITY(1,1),
    UrunId INT NOT NULL,
    DepoId INT NOT NULL,
    Miktar INT NOT NULL DEFAULT 0,
    OlusturulmaTarihi DATETIME DEFAULT GETDATE(),
    GuncellemeTarihi DATETIME DEFAULT GETDATE(),
    CONSTRAINT FK_Stok_Urun FOREIGN KEY (UrunId) REFERENCES Urun(UrunId),
    CONSTRAINT FK_Stok_Depo FOREIGN KEY (DepoId) REFERENCES Depo(DepoId),
    CONSTRAINT UQ_Stok_Urun_Depo UNIQUE (UrunId, DepoId),
    CONSTRAINT CK_Stok_Miktar CHECK (Miktar >= 0)
);
PRINT 'Stok tablosu olusturuldu.';
GO

-- 8. STOK HAREKETI TABLOSU
CREATE TABLE StokHareketi
(
    HareketId INT PRIMARY KEY IDENTITY(1,1),
    UrunId INT NOT NULL,
    DepoId INT NOT NULL,
    OncekiMiktar INT NOT NULL,
    YeniMiktar INT NOT NULL,
    DegisenMiktar INT NOT NULL,
    CikisMi BIT NOT NULL,
    HareketTipi NVARCHAR(20) NOT NULL,
    Aciklama NVARCHAR(500) NULL,
    KullaniciAdi NVARCHAR(100) NULL,
    HareketTarihi DATETIME DEFAULT GETDATE(),
    CONSTRAINT FK_StokHareketi_Urun FOREIGN KEY (UrunId) REFERENCES Urun(UrunId),
    CONSTRAINT FK_StokHareketi_Depo FOREIGN KEY (DepoId) REFERENCES Depo(DepoId),
    CONSTRAINT CK_StokHareketi_CikisMi CHECK ((CikisMi = 0 AND DegisenMiktar >= 0) OR (CikisMi = 1 AND DegisenMiktar < 0))
);
PRINT 'StokHareketi tablosu olusturuldu.';
GO

-- 10. STORED PROCEDURE - BARKODA GORE URUN GETIR
IF OBJECT_ID('sp_GetUrunByBarkod') IS NOT NULL
    DROP PROCEDURE sp_GetUrunByBarkod;
GO

CREATE PROCEDURE sp_GetUrunByBarkod
    @Barkod NVARCHAR(50)
AS
BEGIN
    SELECT TOP 1
        u.UrunId,
        u.Barkod,
        u.UrunAdi,
        u.KategoriId,
        k.KategoriAdi,
        u.BirimId,
        b.BirimAdi,
        u.Fiyat,
        u.MinimumStok,
        u.MaksimumStok,
        u.AktifMi
    FROM Urun u
    INNER JOIN Kategori k ON u.KategoriId = k.KategoriId
    INNER JOIN Birim b ON u.BirimId = b.BirimId
    WHERE u.Barkod = @Barkod AND u.AktifMi = 1;
END
GO

-- 11. STORED PROCEDURE - TUM URUNLERI GETIR
IF OBJECT_ID('sp_GetAllUrunler') IS NOT NULL
    DROP PROCEDURE sp_GetAllUrunler;
GO

CREATE PROCEDURE sp_GetAllUrunler
AS
BEGIN
    SELECT
        u.UrunId,
        u.Barkod,
        u.UrunAdi,
        k.KategoriAdi,
        b.BirimAdi,
        u.Fiyat,
        u.MinimumStok,
        u.MaksimumStok,
        u.AktifMi,
        u.OlusturulmaTarihi
    FROM Urun u
    INNER JOIN Kategori k ON u.KategoriId = k.KategoriId
    INNER JOIN Birim b ON u.BirimId = b.BirimId
    WHERE u.AktifMi = 1
    ORDER BY u.UrunAdi;
END
GO

-- 12. STORED PROCEDURE - DEPO STOKU GETIR
IF OBJECT_ID('sp_GetStokByDepo') IS NOT NULL
    DROP PROCEDURE sp_GetStokByDepo;
GO

CREATE PROCEDURE sp_GetStokByDepo
    @DepoId INT
AS
BEGIN
    SELECT
        s.StokId,
        s.UrunId,
        u.Barkod,
        u.UrunAdi,
        k.KategoriAdi,
        b.BirimAdi,
        u.Fiyat,
        s.Miktar,
        CASE 
            WHEN s.Miktar <= u.MinimumStok THEN 'Kritik'
            WHEN s.Miktar <= (u.MaksimumStok / 4) THEN 'Dusuk'
            ELSE 'Normal'
        END AS StokDurumu,
        s.OlusturulmaTarihi,
        s.GuncellemeTarihi
    FROM Stok s
    INNER JOIN Urun u ON s.UrunId = u.UrunId
    INNER JOIN Kategori k ON u.KategoriId = k.KategoriId
    INNER JOIN Birim b ON u.BirimId = b.BirimId
    WHERE s.DepoId = @DepoId AND u.AktifMi = 1
    ORDER BY u.UrunAdi;
END
GO

-- 13. STORED PROCEDURE - STOK EKLE/GUNCELLE
IF OBJECT_ID('sp_UpsertStok') IS NOT NULL
    DROP PROCEDURE sp_UpsertStok;
GO

CREATE PROCEDURE sp_UpsertStok
    @UrunId INT,
    @DepoId INT,
    @Miktar INT
AS
BEGIN
    IF EXISTS(SELECT 1 FROM Stok WHERE UrunId = @UrunId AND DepoId = @DepoId)
    BEGIN
        UPDATE Stok
        SET Miktar = @Miktar,
            GuncellemeTarihi = GETDATE()
        WHERE UrunId = @UrunId AND DepoId = @DepoId;
    END
    ELSE
    BEGIN
        INSERT INTO Stok (UrunId, DepoId, Miktar)
        VALUES (@UrunId, @DepoId, @Miktar);
    END
END
GO

PRINT 'Stored Procedures olusturuldu.';
